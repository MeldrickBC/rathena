//===== Script =======================================
//= Card Removal NPC
//====================================================
-	script	Card Remover	-1,{
	disable_items;
	mes "[Card Remover Lady]";
	mes "Good day! I have the power to remove cards that you have compounded onto your equipment, free of charge! Are you interested?";
	next;
	switch(select("Yes.:No thanks.")) {
	case 1:
		mes "[Card Remover Lady]";
		mes "Very well. What item shall I examine for you?";
		next;
		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Card Remover Lady]";
			mes "Um... You are not wearing anything that I can remove cards from.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Card Remover Lady]";
			mes "Listen... There are no cards compounded on this item. I can do nothing with it, I'm afraid.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355FFJust a minute!";
			mes "I can't offer any of my";
			mes "services to you because";
			mes "you're carrying too much";
			mes "stuff. Put your extra items in";
			mes "Kafra Storage and come again~";
			close;
		}
		if(.@cardcount > 1) {
			.@plural$ = "s";
		} 
		else {
			.@plural$ = "";
		}
		mes "[Card Remover Lady]";
		mes "This item contains " + .@cardcount + " card"+.@plural$+" compounded on it. Shall we continue?";
		next;
		if(select("Yes, do it.:Never mind.") == 2) {
			mes "[Card Remover Lady]";
			mes "Very well. Return at once if you seek my services.";
			close;
		}		
		mes "[Card Remover Lady]";
		mes "Very well. Let's begin.";			
		next;
		mes "[Card Remover Lady]";
		mes "...";
		specialeffect EF_SHRINK;
		specialeffect EF_ENTRY;			
		next;
		successremovecards .@part;
		specialeffect EF_HIT6;
		mes "[Card Remover Lady]";
		mes "Here are your items. Until next time!";
		close;
	case 2:
		mes "[Card Remover Lady]";
		mes "Very well. Return at once if you seek my services.";
		close;
	}
}
prontera,171,226,3	duplicate(Card Remover)	Card Remover#prontera	10590
