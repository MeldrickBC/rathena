	if (checkquest(100000) == -1) {
		mes "Do you want to activate killing quest?";
			next;
			switch(select("Yes:No")) {
			case 1:
				mes "[Cuir]";
				setquest 100000;
				mes "Initiating Quest.";				
				close;				
			case 2:
				mes "[Cuir]";
				mes "Ok.";
				close;

	}

	set .@baselevel,BaseLevel;
	set .moblevel,getmonsterinfo( killedrid,MOB_LV );
	
	set .leveldiff, .moblevel - .@baselevel;
	if (.leveldiff <= 10 && .leveldiff >= -10 && .@baselevel >= 10 && .@baselevel <= 99) {		
		if (rand(1, 100) <= .@baselevel + .leveldiff) {
			getitem 2999000,1;
			set .@aid,getcharid(0);
			set .itemname$,getitemname(2999000);
			set .countitem,countitem(2999000);
			if (.countitem > 19) {
				if  (.@baselevel >= 10 && .@baselevel <= 19) {
					getitem 100342,1;
				}
				else if (.@baselevel >= 20 && .@baselevel <= 29) {
					getitem 100342,2;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 39) {
					getitem 100343,1;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 49) {
					getitem 100343,2;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 59) {
					getitem 100343,3;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 69) {
					getitem 100344,1;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 79) {
					getitem 100344,2;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 89) {
					getitem 100344,3;
				}
				else if (.@baselevel >= 10 && .@baselevel <= 99) {
					getitem 100345,1;
				}
				delitem 2999000,20;
			}
		}
	}
	else if (.@baselevel == 100) {
		set .droprate,0;
		set .qty,1;
		if (.moblevel >= 10 && .moblevel <= 50) {
			.droprate = 1;			
		}
		else if (.moblevel >= 51 && .moblevel <= 80) {
			.droprate = 5;
		}
		else if (.moblevel >= 81 && .moblevel <= 100) {
			.droprate = 20;
		}
		else if (.moblevel > 100) {
			.droprate = 30;
			.qty = 3;
		}
		if (rand(1,100) <= .droprate) {
			getitem 2999000,rand(1,.qty);
		}
	}
}
